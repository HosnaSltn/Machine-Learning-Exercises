\section{تحلیل مرز تصمیم‌گیری غیرخطی و تابع هزینهٔ \(L_2\)}

\noindent\textbf{سوال}
داده‌های کلاس \(1\) درون دایرهٔ واحد به مرکز مبدأ و داده‌های کلاس \(0\) بیرون آن قرار دارند. مدل رگرسیون لجستیک با ویژگی‌های درجهٔ دو به صورت زیر تعریف می‌شود.
\begin{align*}
    \phi(x)&=\begin{bmatrix}1&x_1&x_2&x_1^2&x_2^2\end{bmatrix}^{\top},

\end{align*}

\subsection*{الف) استنتاج پارامترهای مرز دایره‌ای}
\begin{questionprompt}
    برداری برای \(\theta=[\theta_0,\theta_1,\theta_2,\theta_3,\theta_4]^\top\) پیدا کنید که مرز تصمیم آن دایره‌ای به مرکز \((0,0)\) و شعاع \(1\) باشد. از معادلهٔ مرز تصمیم و معادلهٔ دایره استفاده کنید.
\end{questionprompt}
می‌دانیم معادلهٔ مرز تصمیم‌گیری به صورت زیر است.
\[
    \theta^\top\phi(x)=0
    \quad\Longleftrightarrow\quad
    \theta_0+\theta_1x_1+\theta_2x_2+\theta_3x_1^2+\theta_4x_2^2=0.
\]
با توجه به شکل، مرز تصمیم‌گیری دایره‌ای به مرکز \((0,0)\) و شعاع \(1\) است. بنابراین
\[
    x_1^2+x_2^2=1
    \quad\Longleftrightarrow\quad
    x_1^2+x_2^2-1=0.
\]
با مقایسهٔ ضرایب، بردار پارامترها به دست می‌آید.
\[
    \theta=\begin{bmatrix}-1&0&0&1&1\end{bmatrix}^{\top}.
\]

\subsection*{ب) یکتا بودن پارامترها}
\begin{questionprompt}
    آیا بردار پارامترهای قسمت قبل برای ایجاد مرز دایره‌ای یکتا است؟ اگر خیر، یک بردار دیگر مثال بزنید و دلیل را توضیح دهید.
\end{questionprompt}
خیر، بردار پارامترها یکتا نیست. اگر \(\theta^\top\phi(x)=0\) باشد، برای هر ضریب \(\alpha\neq0\) داریم
\[
    (\alpha\theta)^\top\phi(x)
      =\alpha\theta^\top\phi(x)=0.
\]
\[
    \alpha\theta
      =\begin{bmatrix}-\alpha&0&0&\alpha&\alpha\end{bmatrix}^{\top}
\]
همان مرز را ایجاد می‌کند. برای مثال، با \(\alpha=2\) داریم
\[
    2\theta=\begin{bmatrix}-2&0&0&2&2\end{bmatrix}^{\top}.
\]

\subsection*{ج) تابع هزینه با تنظیم‌سازی \(L_2\)}
\begin{questionprompt}
    تابع هزینه را برای \(m\) نمونهٔ آموزشی با تنظیم‌سازی \(L_2\) بنویسید. پارامتر بایاس \(\theta_0\) در جریمه لحاظ نمی‌شود.
\end{questionprompt}
تابع هزینه برابر است با
\[
    J(\theta)=\frac1m\sum_{i=1}^{m}
      \operatorname{cost}\bigl(x^{(i)},y^{(i)}\bigr)+\lambda R(\theta).
\]
با قرار دادن \(p_i=h_\theta(x^{(i)})\)، تابع هزینهٔ رگرسیون لجستیک با تنظیم‌سازی \(L_2\) به صورت زیر است.
\[
    J(\theta)
      =\frac1m\sum_{i=1}^{m}
        \bigl[-y^{(i)}\log p_i-(1-y^{(i)})\log(1-p_i)\bigr]
        +\lambda\sum_{j=1}^{4}\theta_j^2.
\]
جمع جریمه از \(j=1\) آغاز می‌شود تا \(\theta_0\) در آن وارد نشود.

\subsection*{د) مشتق جزئی تابع هزینه}
\begin{questionprompt}
    با استفاده از تابع هزینهٔ بخش (ج)، مشتق جزئی \(\partial J(\theta)/\partial\theta_j\) را برای \(j=1,2,3,4\) محاسبه کنید.
\end{questionprompt}
برای نمونهٔ \(i\)، امتیاز مدل \(z_i=\theta^\top\phi(x^{(i)})\) و خروجی آن \(h_\theta(x^{(i)})=g(z_i)\) است. در این روابط داریم \(\theta_j=[0,\theta_1,\theta_2,\theta_3,\theta_4]^\top\) زیرا بایاس در جریمه وارد نمی‌شود.
\[
    g'(z)=\frac{e^{-z}}{(1+e^{-z})^2}=g(z)\bigl(1-g(z)\bigr).
\]
\[
    \nabla_\theta h_\theta(x^{(i)})=\frac{\partial h_\theta}{\partial z_i}\frac{\partial z_i}{\partial \theta}
      =\phi(x^{(i)})\,h_\theta(x^{(i)})
          \bigl(1-h_\theta(x^{(i)})\bigr).
\]
\begin{align*}
    \nabla_\theta J(\theta)
      &=\frac{1}{m}\sum_{i=1}^{m}\Biggl[
         -y^{(i)}
         \frac{\phi(x^{(i)})\,h_\theta(x^{(i)})
               \bigl(1-h_\theta(x^{(i)})\bigr)}
              {h_\theta(x^{(i)})}\\
      &\qquad -(1-y^{(i)})
         \frac{-\phi(x^{(i)})\,h_\theta(x^{(i)})
                \bigl(1-h_\theta(x^{(i)})\bigr)}
              {1-h_\theta(x^{(i)})}
         \Biggr]+2\lambda \sum_{j=1}^{m}\theta_j,\\[0.5em]
    \nabla_\theta J(\theta)
      &=\frac{1}{m}\sum_{i=1}^{m}\phi(x^{(i)})
         \bigl[-y^{(i)}\bigl(1-h_\theta(x^{(i)})\bigr)
           +(1-y^{(i)})h_\theta(x^{(i)})\bigr]
         +2\lambda \sum_{j=1}^{m}\theta_j,\\[0.5em]
    \nabla_\theta J(\theta)
      &=\frac{1}{m}\sum_{i=1}^{m}\phi(x^{(i)})
         \bigl[-y^{(i)}+h_\theta(x^{(i)})\bigr]
         +2\lambda \sum_{j=1}^{m}\theta_j,\\[0.5em]
    \frac{\partial J(\theta)}{\partial\theta_j}
      &=\frac{1}{m}\sum_{i=1}^{m}\phi_j(x^{(i)})
         \bigl(h_\theta(x^{(i)})-y^{(i)}\bigr)
         +2\lambda\theta_j,\qquad j=1,2,3,4.
\end{align*}

\subsection*{هـ) به‌روزرسانی گرادیان کاهشی}
\begin{questionprompt}
    قاعدهٔ به‌روزرسانی گرادیان کاهشی را برای \(\theta_3\)، ضریب \(x_1^2\)، بنویسید.
\end{questionprompt}
قاعدهٔ گرادیان کاهشی با نرخ یادگیری \(\alpha\) به صورت
\[
    \theta_j^{new}=\theta_j-\alpha\frac{\partial J}{\partial\theta_j}
\]
است. چون \(\theta_3\) ضریب \(x_1^2\) است، داریم
\begin{align*}
    \theta_3^{\mathrm{new}}
      &=\theta_3-\alpha\left[
         \frac1m\sum_{i=1}^{m}(x_1^{(i)})^2
          \bigl(h_\theta(x^{(i)})-y^{(i)}\bigr)+2\lambda\theta_3
         \right]\\
      &=\theta_3(1-2\alpha\lambda)
         -\frac{\alpha}{m}\sum_{i=1}^{m}(x_1^{(i)})^2
           \bigl(h_\theta(x^{(i)})-y^{(i)}\bigr).
\end{align*}

\section{تحلیل تعمیم‌پذیری و مقایسهٔ تنظیم‌سازی \(L_1\) و \(L_2\)}

\noindent\textbf{سوال}
دو مدل رگرسیون لجستیک \(A\) و \(B\) روی داده‌های آموزشی دقت \(100\%\) و مرز مشترک \(x_1+x_2=1\) دارند. بردارهای پارامترهای آن‌ها عبارت‌اند از
\[
    \theta_A=\begin{bmatrix}-3&3&3\end{bmatrix}^{\top}
\]
\[
  \theta_B=\begin{bmatrix}-100&100&100\end{bmatrix}^{\top}
\]

\subsection*{الف) تعمیم‌پذیری و اطمینان بیش از حد}
\begin{questionprompt}
    رفتار خروجی هر مدل را توضیح دهید. کدام مدل خروجی‌هایی نزدیک‌تر به \(0\) و \(1\) تولید می‌کند؟ دربارهٔ تعمیم‌پذیری روی داده‌های جدید استدلال کنید.
\end{questionprompt}
مدل \(B\) خروجی‌های قاطعانه‌تری تولید می‌کند. به دلیل بزرگ‌تر بودن ضرایب، تغییرات خروجی سیگموید نسبت به ورودی در نزدیکی مرز شدیدتر است و خروجی مدل سریع‌تر به \(0\) یا \(1\) نزدیک می‌شود.

انتظار می‌رود مدل \(A\) روی داده‌های جدید تعمیم‌پذیری بهتری داشته باشد، زیرا ضرایب کوچک‌تر هستند و در نتیجه با تغییر اندک ورودی، احتمال پیش‌بینی‌شده نیز کمتر تغییر می‌کند (تابع سیگموید شیب کمتری دارد). در مدل \(B\)، تغییر کوچک در ورودیِ نزدیک به مرز می‌تواند خروجی را به‌شدت تغییر دهد و مدل را نسبت به پیش‌بینی خود بیش از حد مطمئن کند.
\\
\subsection*{ب) کنترل بیش‌برازش}
\begin{questionprompt}
    توضیح دهید افزودن جریمهٔ \(L_1\) یا \(L_2\) چگونه بزرگی ضرایب و اطمینان بیش از حد مدل را کنترل می‌کند.
\end{questionprompt}
اضافه کردن جریمهٔ تنظیم‌سازی باعث می‌شود علاوه بر مینیمم کردن خطا داده‌های آموزشی، بزرگی ضرایب ویژگی‌ها \(\theta\) نیز در تابع هزینه لحاظ می‌شود.
\[
    J(\theta)=\frac1m\sum_{i=1}^{m}
      \operatorname{cost}\bigl(x^{(i)},y^{(i)}\bigr)+\lambda R(\theta).
\]
این جریمه باعث کوچک‌تر شدن ضرایب و ملایم‌تر شدن تغییر خروجی مدل در نزدیکی مرز تصمیم‌گیری می‌شود. ممکن است خطا روی داده‌های آموزشی کمی افزایش پیدا کند، اما این اثر می‌تواند خطا روی داده‌های جدید را کاهش دهد. پارامتر \(\lambda\) شدت جریمه را تعیین می‌کند. با انتخاب مقدار مناسب برای آن، مدل با ضرایب کوچک‌تر، مانند مدل \(A\)، می‌تواند ترجیح داده شود.
\\

\subsection*{ج) مقایسهٔ \(L_1\) و \(L_2\) و انتخاب ویژگی}
\begin{questionprompt}
    تفاوت اثر \(L_1\) و \(L_2\) بر ضرایب نهایی را توضیح دهید. کدام روش می‌تواند به انتخاب ویژگی منجر شود و چرا؟
\end{questionprompt}
در تنظیم‌سازی \(L_2\)، مقدار ضرایب به سمت صفر می‌رود، اما به‌ندرت دقیقاً صفر می‌شود. در تنظیم‌سازی \(L_1\)، ممکن است ضریب ویژگی‌های کم‌اثر دقیقاً صفر شود. با کنار گذاشتن بایاس از جریمه، اگر \(w=[\theta_1,\ldots,\theta_n]^\top\) باشد، داریم
\begin{align*}
    R_{L_2}(\theta)&=\sum_{j=1}^{n}\theta_j^2=\lVert w\rVert_2^2,\\
    R_{L_1}(\theta)&=\sum_{j=1}^{n}|\theta_j|=\lVert w\rVert_1.
\end{align*}
در فضای دو ضریب، خطوط تراز جریمهٔ \(L_2\) دایره‌ای و هموارند. خطوط تراز جریمهٔ \(L_1\) به شکل مربعِ چرخیده‌اند و رأس‌های آن‌ها روی محورهای مختصات قرار دارند. اگر جواب بهینه روی یکی از این رأس‌ها قرار بگیرد، یکی از ضرایب دقیقاً صفر است و ویژگی متناظر از مدل حذف می‌شود. به همین دلیل \(L_1\) می‌تواند به انتخاب ویژگی منجر شود.
