\section{تحلیل شبکهٔ عصبی دولایه با \lr{ReLU} و \lr{Softmax}}

\noindent\textbf{سوال}
یک شبکهٔ عصبی دولایه برای دسته‌بندی سه‌کلاسه در نظر بگیرید. ورودی \(x=\begin{bmatrix}x_1&x_2\end{bmatrix}\) دو ویژگی دارد. لایهٔ مخفی شامل دو نورون \(h_1\) و \(h_2\) با تابع فعال‌سازی \(\ReLU(z)=\max(0,z)\) است. لایهٔ خروجی سه نورون \(o_1\)، \(o_2\) و \(o_3\)، یکی برای هر کلاس، دارد و از \lr{Softmax} برای تولید احتمال‌ها استفاده می‌کند.
\[
    W^{(1)}=\begin{bmatrix}1&-2\\3&-1\end{bmatrix},
    \qquad b^{(1)}=\begin{bmatrix}-1&0\end{bmatrix},
    \qquad W^{(2)}=\begin{bmatrix}2&1&-1\\-1&1&3\end{bmatrix},
    \qquad b^{(2)}=\begin{bmatrix}0&1&0\end{bmatrix}.
\]

\subsection*{الف) معادلات انتشار پیشرو}
\begin{questionprompt}
    مجموعهٔ کامل معادلات انتشار پیشرو را از ورودی \(x\) تا بردار احتمال نهایی \(p\) بنویسید. معادلات \(z^{(1)}\) (خروجی خطی لایهٔ مخفی)، \(a^{(1)}\) (خروجی فعال‌شدهٔ لایهٔ مخفی)، \(z^{(2)}\) (امتیازهای خروجی) و \(p\) (احتمال‌های \lr{Softmax}) را ارائه دهید.
\end{questionprompt}
بردارها را سطری در نظر می‌گیریم. ابتدا ورودی خطی لایهٔ مخفی را محاسبه می‌کنیم و تابع \lr{ReLU} را روی هر مؤلفه اعمال می‌کنیم. سپس امتیازهای لایهٔ خروجی و احتمال هر کلاس را به دست می‌آوریم.
\begin{align*}
    z^{(1)} &= xW^{(1)}+b^{(1)}
              = \begin{bmatrix}x_1+3x_2-1 & -2x_1-x_2\end{bmatrix},\\
    a^{(1)} &= \ReLU(z^{(1)}), \qquad
               a_j^{(1)}=\max(0,z_j^{(1)}) \quad (j=1,2),\\
    z^{(2)} &= a^{(1)}W^{(2)}+b^{(2)},\\
    p &= \softmax(z^{(2)}), \qquad
         p_k=\frac{e^{z_k^{(2)}}}{\sum_{j=1}^{3}e^{z_j^{(2)}}}
         \quad (k=1,2,3).
\end{align*}

\subsection*{ب) محاسبهٔ پیش‌بینی}
\begin{questionprompt}
    برای \(x=\begin{bmatrix}2&1\end{bmatrix}\)، گام‌به‌گام خروجی خطی لایهٔ مخفی \(z^{(1)}\)، خروجی فعال‌شدهٔ آن \(a^{(1)}\)، امتیازهای نهایی \(z^{(2)}\) و بردار احتمال \(p\) را پس از اعمال \lr{Softmax} به دست آورید. شبکه کدام کلاس را پیش‌بینی می‌کند؟
\end{questionprompt}
برای \(x=\begin{bmatrix}2&1\end{bmatrix}\)، خروجی خطی و فعال‌شدهٔ لایهٔ مخفی و امتیازهای خروجی به‌ترتیب برابرند با
\begin{align*}
    z^{(1)}
    &=\begin{bmatrix}2&1\end{bmatrix}\begin{bmatrix}1&-2\\3&-1\end{bmatrix}
      +\begin{bmatrix}-1&0\end{bmatrix}
     =\begin{bmatrix}5&-5\end{bmatrix}+\begin{bmatrix}-1&0\end{bmatrix}
     =\begin{bmatrix}4&-5\end{bmatrix},\\
    a^{(1)}
    &=\ReLU\!\left(\begin{bmatrix}4&-5\end{bmatrix}\right)
     =\begin{bmatrix}4&0\end{bmatrix},\\
    z^{(2)}
    &=\begin{bmatrix}4&0\end{bmatrix}\begin{bmatrix}2&1&-1\\-1&1&3\end{bmatrix}
      +\begin{bmatrix}0&1&0\end{bmatrix}
     =\begin{bmatrix}8&4&-4\end{bmatrix}+\begin{bmatrix}0&1&0\end{bmatrix}
     =\begin{bmatrix}8&5&-4\end{bmatrix},\\
    p
    &=\begin{bmatrix}
        \dfrac{e^8}{e^8+e^5+e^{-4}} &
        \dfrac{e^5}{e^8+e^5+e^{-4}} &
        \dfrac{e^{-4}}{e^8+e^5+e^{-4}}
    \end{bmatrix}\\
    &\approx\begin{bmatrix}0.95256855&0.04742560&5.85278346\times10^{-6}\end{bmatrix}.
\end{align*}
بیشترین احتمال به کلاس اول تعلق دارد، بنابراین \(\hat{y}=\argmax_k p_k=1\).

\subsection*{ج) محاسبهٔ خطای آنتروپی متقاطع}
\begin{questionprompt}
    فرض کنید کلاس واقعی ورودی \(x=\begin{bmatrix}2&1\end{bmatrix}\) کلاس دوم است و بردار \lr{one-hot} آن \(y=\begin{bmatrix}0&1&0\end{bmatrix}\) است. با استفاده از احتمال‌های بخش (ب) و فرمول \(L_i=-\sum_{k=1}^{C} y_k\log(p_k)\)، خطای آنتروپی متقاطع دسته‌ای این نمونه را محاسبه کنید.
\end{questionprompt}
برچسب واقعی کلاس دوم است و \(y=\begin{bmatrix}0&1&0\end{bmatrix}\).
در مجموع آنتروپی متقاطع، تنها جملهٔ مربوط به کلاس دوم باقی می‌ماند.
\begin{align*}
    L_i
    &=-\sum_{k=1}^{3}y_k\ln p_k=-\ln p_2\\
    &=-\ln\!\left(\frac{e^5}{e^8+e^5+e^{-4}}\right)
     =\ln(e^8+e^5+e^{-4})-5\approx3.048593.
\end{align*}

\subsection*{د) محاسبهٔ گرادیان و پس‌انتشار}
\begin{questionprompt}
    با استفاده از نتایج بخش‌های (ب) و (ج)، یک گام کامل پس‌انتشار برای خطای \(L_i\) انجام دهید.
    \begin{enumerate}
        \item گرادیان اولیهٔ \(\partial L_i/\partial z^{(2)}=p-y\) را محاسبه کنید.
        \item با قاعدهٔ زنجیره‌ای، گرادیان‌های \(W^{(2)}\) و \(b^{(2)}\) را به دست آورید.
        \item گرادیان را به لایهٔ مخفی پس‌انتشار دهید و \(\partial L_i/\partial a^{(1)}\) و \(\partial L_i/\partial z^{(1)}\) را با توجه به مشتق \lr{ReLU} محاسبه کنید.
        \item گرادیان‌های \(W^{(1)}\) و \(b^{(1)}\) را محاسبه کنید.
    \end{enumerate}
\end{questionprompt}
\textbf{۱. گرادیان نسبت به امتیازهای خروجی.}
ترکیب \lr{Softmax} و خطای آنتروپی متقاطع می‌دهد
\begin{align*}
    \delta^{(2)}=\frac{\partial L_i}{\partial z^{(2)}}
    &=p-y=\begin{bmatrix}p_1 & p_2-1 & p_3\end{bmatrix}\\
    &\approx\begin{bmatrix}0.95256855&-0.95257440&5.85278346\times10^{-6}\end{bmatrix}.
\end{align*}

\noindent\textbf{۲. گرادیان پارامترهای لایهٔ دوم.}
هر وزن \(W_{rs}^{(2)}\) و بایاس \(b_s^{(2)}\) فقط بر امتیاز \(z_s^{(2)}\) اثر می‌گذارند. قاعدهٔ زنجیره‌ای برای هر مؤلفه به صورت زیر است.
\begin{align*}
    \frac{\partial L_i}{\partial W_{rs}^{(2)}}
    &=\frac{\partial L_i}{\partial z_s^{(2)}}
      \frac{\partial z_s^{(2)}}{\partial W_{rs}^{(2)}}
      =\delta_s^{(2)}a_r^{(1)},\\
    \frac{\partial L_i}{\partial b_s^{(2)}}
    &=\frac{\partial L_i}{\partial z_s^{(2)}}
      \frac{\partial z_s^{(2)}}{\partial b_s^{(2)}}
      =\delta_s^{(2)}.
\end{align*}
با کنار هم قرار دادن مؤلفه‌ها، گرادیان وزن‌ها به صورت حاصل‌ضرب بیرونی به دست می‌آید.
\begin{align*}
    \frac{\partial L_i}{\partial W^{(2)}}
    &=(a^{(1)})^\top\delta^{(2)}
      =\begin{bmatrix}4\\0\end{bmatrix}\delta^{(2)}\\
    &\approx\begin{bmatrix}3.81027421&-3.81029762&2.34111338\times10^{-5}\\0&0&0\end{bmatrix},\\
    \frac{\partial L_i}{\partial b^{(2)}}
    &=\delta^{(2)}
      \approx\begin{bmatrix}0.95256855&-0.95257440&5.85278346\times10^{-6}\end{bmatrix}.
\end{align*}

\noindent\textbf{۳. پس‌انتشار به لایهٔ مخفی.}
مشتق \lr{ReLU} برای ورودی مثبت یک و برای ورودی منفی صفر است. در این نمونه \(z^{(1)}=\begin{bmatrix}4&-5\end{bmatrix}\)، پس بردار مشتق برابر \(\begin{bmatrix}1&0\end{bmatrix}\) است. نماد \(\odot\) ضرب مؤلفه‌به‌مؤلفه را نشان می‌دهد.
\begin{align*}
    \frac{\partial L_i}{\partial a^{(1)}}
    &=\frac{\partial L_i}{\partial z^{(2)}}
      \frac{\partial z^{(2)}}{\partial a^{(1)}}
      =\delta^{(2)}(W^{(2)})^\top\\
    &=\delta^{(2)}\begin{bmatrix}2&-1\\1&1\\-1&3\end{bmatrix}
      \approx\begin{bmatrix}0.95255685&-1.90512540\end{bmatrix},\\
    \delta^{(1)}=\frac{\partial L_i}{\partial z^{(1)}}
    &=\frac{\partial L_i}{\partial a^{(1)}}
      \frac{\partial a^{(1)}}{\partial z^{(1)}}
      =\frac{\partial L_i}{\partial a^{(1)}}\odot\ReLU'(z^{(1)})\\
    &=\frac{\partial L_i}{\partial a^{(1)}}\odot\begin{bmatrix}1&0\end{bmatrix}
      \approx\begin{bmatrix}0.95255685&0\end{bmatrix}.
\end{align*}

\noindent\textbf{۴. گرادیان پارامترهای لایهٔ اول.}
وزن \(W_{rs}^{(1)}\) و بایاس \(b_s^{(1)}\) فقط در محاسبهٔ \(z_s^{(1)}\) وارد می‌شوند. بنابراین
\begin{align*}
    \frac{\partial L_i}{\partial W_{rs}^{(1)}}
    &=\frac{\partial L_i}{\partial z_s^{(1)}}
      \frac{\partial z_s^{(1)}}{\partial W_{rs}^{(1)}}
      =\delta_s^{(1)}x_r,\\
    \frac{\partial L_i}{\partial b_s^{(1)}}
    &=\frac{\partial L_i}{\partial z_s^{(1)}}
      \frac{\partial z_s^{(1)}}{\partial b_s^{(1)}}
      =\delta_s^{(1)}.
\end{align*}
در قالب ماتریس و بردار داریم
\begin{align*}
    \frac{\partial L_i}{\partial W^{(1)}}
    &=x^\top\delta^{(1)}
     =\begin{bmatrix}2\\1\end{bmatrix}\delta^{(1)}
     \approx\begin{bmatrix}1.90511369&0\\0.95255685&0\end{bmatrix},\\
    \frac{\partial L_i}{\partial b^{(1)}}&=\delta^{(1)}
     \approx\begin{bmatrix}0.95255685&0\end{bmatrix}.
\end{align*}

\subsection*{هـ) به‌روزرسانی پارامترها}
\begin{questionprompt}
    با نرخ یادگیری \(\alpha=0.1\) و گرادیان‌های بخش (د)، مقادیر جدید \(W^{(1)}\)، \(b^{(1)}\)، \(W^{(2)}\) و \(b^{(2)}\) را پس از یک گام گرادیان کاهشی محاسبه کنید. از رابطهٔ \(\theta_{\mathrm{new}}=\theta_{\mathrm{old}}-\alpha\,\partial L_i/\partial\theta\) استفاده کنید.
\end{questionprompt}
با نرخ یادگیری \(\alpha=0.1\)، همهٔ پارامترها را با گرادیان‌های محاسبه‌شده در نقطهٔ اولیه و طبق رابطهٔ زیر به‌روزرسانی می‌کنیم.
\[
    \theta_{\mathrm{new}}=\theta_{\mathrm{old}}-0.1\,\frac{\partial L_i}{\partial\theta}.
\]
\begin{align*}
    W_{\mathrm{new}}^{(1)}
    &=W^{(1)}-0.1\,\frac{\partial L_i}{\partial W^{(1)}}
      =\begin{bmatrix}1&-2\\3&-1\end{bmatrix}-0.1\,x^\top\delta^{(1)}\\
    &\approx\begin{bmatrix}0.80948863&-2\\2.90474432&-1\end{bmatrix},\\
    b_{\mathrm{new}}^{(1)}
    &=b^{(1)}-0.1\,\frac{\partial L_i}{\partial b^{(1)}}
      =\begin{bmatrix}-1&0\end{bmatrix}-0.1\,\delta^{(1)}\\
    &\approx\begin{bmatrix}-1.09525568&0\end{bmatrix},\\
    W_{\mathrm{new}}^{(2)}
    &=W^{(2)}-0.1\,\frac{\partial L_i}{\partial W^{(2)}}
      =\begin{bmatrix}2&1&-1\\-1&1&3\end{bmatrix}-0.1\,(a^{(1)})^\top\delta^{(2)}\\
    &\approx\begin{bmatrix}1.61897258&1.38102976&-1.00000234\\-1&1&3\end{bmatrix},\\
    b_{\mathrm{new}}^{(2)}
    &=b^{(2)}-0.1\,\frac{\partial L_i}{\partial b^{(2)}}
      =\begin{bmatrix}0&1&0\end{bmatrix}-0.1\,\delta^{(2)}\\
    &\approx\begin{bmatrix}-0.09525686&1.09525744&-5.85278346\times10^{-7}\end{bmatrix}.
\end{align*}

\subsection*{و) تحلیل گرادیان نورون دوم لایهٔ مخفی}
\begin{questionprompt}
    با توجه به نتایج پس‌انتشار، به پرسش‌های زیر دربارهٔ نورون دوم لایهٔ مخفی پاسخ دهید.
    \begin{enumerate}
        \item مقدار \(\partial L_i/\partial z_2^{(1)}\) چقدر است؟
        \item با رجوع به نتایج بند ۳ بخش (د) و بند ۲ بخش (ب)، توضیح دهید این گرادیان چرا صفر می‌شود. به مشتق تابع و مقدار ورودی آن اشاره کنید.
        \item در گام به‌روزرسانی بخش (هـ)، چه اتفاقی برای وزن‌های \(W_{1,2}^{(1)}\) و \(W_{2,2}^{(1)}\) و بایاس \(b_2^{(1)}\) افتاد؟ این نتیجه برای یادگیری نورون دوم از این ورودی چه مفهومی دارد؟
    \end{enumerate}
\end{questionprompt}
\textbf{۱. مقدار گرادیان.}
برای ورودی خطی نورون دوم داریم \(\frac{\partial L_i}{\partial z_2^{(1)}}=0\).

\noindent\textbf{۲. علت صفر شدن گرادیان.}
چون \(z_2^{(1)}=-5<0\)، خروجی این نورون \(a_2^{(1)}=0\) و مشتق \lr{ReLU} در این نقطه نیز صفر است. قاعدهٔ زنجیره‌ای نتیجه می‌دهد
\[
    \frac{\partial L_i}{\partial z_2^{(1)}}=
    \frac{\partial L_i}{\partial a_2^{(1)}}\,
    \frac{\partial a_2^{(1)}}{\partial z_2^{(1)}}
    =\frac{\partial L_i}{\partial a_2^{(1)}}\,\ReLU'(-5)=0.
\]

\noindent\textbf{۳. اثر بر وزن‌ها و بایاس.}
گرادیان‌های \(W_{1,2}^{(1)}\)، \(W_{2,2}^{(1)}\) و \(b_2^{(1)}\) صفرند و مقادیر آن‌ها بعد از این گام به‌ترتیب \(-2\)، \(-1\) و \(0\) باقی می‌مانند. بنابراین این نمونه برای پارامترهای ورودی نورون دوم سیگنال یادگیری ایجاد نمی‌کند. اگر ورودی خطی نورون برای همهٔ نمونه‌های آموزشی منفی بماند، نورون در وضعیت «مرده» قرار می‌گیرد.
